63static KeyboardModifier keyboardModifierFromString(const QString &modifier)
64{
65 using enum KeyboardModifier;
66 if (modifier.compare(kModifierNameShift, Qt::CaseInsensitive) == 0 ||
67 modifier.localeAwareCompare(QObject::tr("Shift")) == 0)
68 return Shift;
69 if (modifier.compare(kModifierNameCtrl, Qt::CaseInsensitive) == 0 ||
70 modifier.localeAwareCompare(QObject::tr("Ctrl")) == 0)
71 return Ctrl;
72 if (modifier.compare(kModifierNameAlt, Qt::CaseInsensitive) == 0 ||
73 modifier.localeAwareCompare(QObject::tr("Alt")) == 0)
74 return Alt;
75 if (modifier.compare(kModifierNameMeta, Qt::CaseInsensitive) == 0 ||
76 modifier.localeAwareCompare(QObject::tr("Meta")) == 0)
77 return Meta;
78 if (modifier.compare(kModifierNameSuper, Qt::CaseInsensitive) == 0 ||
79 modifier.localeAwareCompare(QObject::tr("Super")) == 0)
80 return Super;
81 if (modifier.compare(kModifierNameAltGr, Qt::CaseInsensitive) == 0 ||
82 modifier.localeAwareCompare(QObject::tr("AltGr")) == 0)
83 return AltGr;
84 if (modifier.compare(kModifierNameNone, Qt::CaseInsensitive) == 0 ||
85 modifier.localeAwareCompare(QObject::tr("None")) == 0)
86 return None;

95static int32_t modifierIDValueFromString(const QString &modifier)
