cmake_minimum_required(VERSION 3.24)
project(DeskflowFirstUse LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOGEN_PARALLEL 1)
find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets Test Network LinguistTools)
get_filename_component(defaultSource "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DESKFLOW_BASE_SOURCE_DIR "${defaultSource}" CACHE PATH "Complete source tree")
set(DESKFLOW_CHANGE_SOURCE_DIR "${DESKFLOW_BASE_SOURCE_DIR}" CACHE PATH "Source overlay")
set(base "${DESKFLOW_BASE_SOURCE_DIR}/src/lib")
set(changes "${DESKFLOW_CHANGE_SOURCE_DIR}/src/lib")
set(sources gui/widgets/ConnectionHome.cpp gui/widgets/ConnectionHome.h gui/IdentityBindings.cpp gui/IdentityBindings.h
  gui/dialogs/CandidatePermissionDialog.cpp gui/dialogs/CandidatePermissionDialog.h
  gui/dialogs/PairingReviewDialog.cpp gui/dialogs/PairingReviewDialog.h gui/dialogs/TrustedDevicesDialog.cpp gui/dialogs/TrustedDevicesDialog.h)
foreach(source IN LISTS sources)
  if(EXISTS "${changes}/${source}")
    list(APPEND actualSources "${changes}/${source}")
  else()
    list(APPEND actualSources "${base}/${source}")
  endif()
endforeach()
add_library(FirstUseComponents STATIC ${actualSources})
# These fake-provider fixtures intentionally exercise the macOS guide branch.
# Native Windows rejection is covered separately without this test-only define.
target_compile_definitions(FirstUseComponents PRIVATE DESKFLOW_PERMISSION_GUIDE_TEST)
target_include_directories(FirstUseComponents PUBLIC "${changes}" "${base}")
target_link_libraries(FirstUseComponents PUBLIC Qt6::Widgets Qt6::Network)
set(ts "${DESKFLOW_BASE_SOURCE_DIR}/translations/deskflow_zh_CN.ts")
if(EXISTS "${DESKFLOW_CHANGE_SOURCE_DIR}/translations/deskflow_zh_CN.ts")
  set(ts "${DESKFLOW_CHANGE_SOURCE_DIR}/translations/deskflow_zh_CN.ts")
endif()
qt_add_lrelease(TS_FILES "${ts}" LRELEASE_TARGET FirstUseTranslations QM_FILES_OUTPUT_VARIABLE qmFiles)
add_executable(HomeUxTests HomeUxTests.cpp)
target_link_libraries(HomeUxTests PRIVATE FirstUseComponents Qt6::Test)
add_dependencies(HomeUxTests FirstUseTranslations)
qt_add_resources(HomeUxTests first_use_translations PREFIX "/translations" BASE "${CMAKE_CURRENT_BINARY_DIR}" FILES ${qmFiles})
enable_testing()
add_test(NAME HomeUxTests COMMAND HomeUxTests)
set_tests_properties(HomeUxTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
get_filename_component(changeRoot "${DESKFLOW_CHANGE_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(baseRoot "${DESKFLOW_BASE_SOURCE_DIR}/.." ABSOLUTE)
set(homeTests "${baseRoot}/native-tests/apple-ui/HomeTests.cpp")
if(EXISTS "${changeRoot}/native-tests/apple-ui/HomeTests.cpp")
  set(homeTests "${changeRoot}/native-tests/apple-ui/HomeTests.cpp")
endif()
add_executable(HomeRegressionTests "${homeTests}")
target_link_libraries(HomeRegressionTests PRIVATE FirstUseComponents Qt6::Test)
add_test(NAME HomeRegressionTests COMMAND HomeRegressionTests)
set_tests_properties(HomeRegressionTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)

# Only Settings is linked for fresh-profile checks. No certificate generation,
# network, core, input injection, service or permission code is linked.
set(CMAKE_PROJECT_PROPER_NAME Deskflow)
set(CMAKE_PROJECT_REV_FQDN org.deskflow.firstusetests)
set(CMAKE_PROJECT_VERSION 1.27.0.10)
set(CMAKE_PROJECT_VERSION_MAJOR 1)
set(CMAKE_PROJECT_VERSION_MINOR 27)
set(CMAKE_PROJECT_VERSION_PATCH 0)
set(CMAKE_PROJECT_VERSION_TWEAK 10)
set(GIT_SHA_SHORT isolated-first-use-tests)
set(CORE_BINARY deskflow-core-disabled)
set(DESKFLOW_RUNTIME_ID deskflow-first-use-tests)
set(REQUIRED_MSVC_RUNTIME_MAJOR 14)
set(REQUIRED_MSVC_RUNTIME_MINOR 0)
configure_file("${base}/common/Constants.h.in" "generated/common/Constants.h" @ONLY)
configure_file("${base}/common/VersionInfo.h.in" "generated/common/VersionInfo.h" @ONLY)
set(settingsSources)
foreach(source Settings.cpp Settings.h LogLevel.h Enums.h QSettingsProxy.cpp IsolatedSession.cpp RuntimeProfile.cpp)
  if(EXISTS "${changes}/common/${source}")
    list(APPEND settingsSources "${changes}/common/${source}")
  else()
    list(APPEND settingsSources "${base}/common/${source}")
  endif()
endforeach()
add_executable(FirstRunSettingsTests FirstRunSettingsTests.cpp ${settingsSources})
target_compile_definitions(FirstRunSettingsTests PRIVATE DESKFLOW_LAB_BUILD)
target_include_directories(FirstRunSettingsTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" "${changes}" "${base}" "${base}/common")
target_link_libraries(FirstRunSettingsTests PRIVATE Qt6::Network Qt6::Test)
add_test(NAME FirstRunSettingsTests COMMAND FirstRunSettingsTests)
set_tests_properties(FirstRunSettingsTests PROPERTIES TIMEOUT 30)
