cmake_minimum_required(VERSION 3.24)
project(DeskflowDisplayRouting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOGEN_PARALLEL 1)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Test)
set(src "${CMAKE_CURRENT_SOURCE_DIR}/../../src/lib")
add_executable(DisplayRoutingTests DisplayRoutingTests.cpp "${src}/common/DisplayTopology.cpp" "${src}/common/NativeDisplayTopology.cpp")
target_include_directories(DisplayRoutingTests PRIVATE "${src}")
target_link_libraries(DisplayRoutingTests PRIVATE Qt6::Gui Qt6::Test)
if(APPLE)
  target_link_libraries(DisplayRoutingTests PRIVATE "-framework CoreGraphics" "-framework ColorSync")
elseif(WIN32)
  target_link_libraries(DisplayRoutingTests PRIVATE user32)
endif()
enable_testing()
add_test(NAME DisplayRoutingTests COMMAND DisplayRoutingTests)
set_tests_properties(DisplayRoutingTests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_executable(DesktopWorkspaceTests DesktopWorkspaceTests.cpp
  "${src}/common/DesktopWorkspace.cpp" "${src}/common/DisplayTopology.cpp")
target_include_directories(DesktopWorkspaceTests PRIVATE "${src}")
target_link_libraries(DesktopWorkspaceTests PRIVATE Qt6::Gui Qt6::Test)
add_test(NAME DesktopWorkspaceTests COMMAND DesktopWorkspaceTests)
set_tests_properties(DesktopWorkspaceTests PROPERTIES TIMEOUT 30)
